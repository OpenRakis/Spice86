            // F000:027B je short 0x0281
            if (!ZeroFlag) {
                // F000:027D mov byte ptr DS:[BX+0x2D],2
                UInt8[DS, (ushort)(BX + 0x2D)] = 2;
            }
            // F000:0281 mov SI,0x0104
            SI = 0x0104;
            // F000:0284 mov DI,0x0106
            DI = 0x0106;
            // F000:0287 mov CX,2
            CX = 2;
            // F000:028A repe cmps byte ptr DS:[SI],byte ptr ES:[DI]
            while (CX != 0) {
                bool equal_028A = UInt8[DS, SI] == UInt8[ES, DI];
                SI = (ushort)(SI + (ushort)State.Direction8);
                DI = (ushort)(DI + (ushort)State.Direction8);
                CX--;
                if (!equal_028A) {
                    break;
                }
            }
            // F000:028C mov byte ptr DS:[BX+0x2A],CL
            UInt8[DS, (ushort)(BX + 0x2A)] = CL;
            // F000:028F pop SI
            SI = Stack.Pop16();
            // F000:0290 pop CX
            CX = Stack.Pop16();
            // F000:0291 mov DX,AX
            DX = AX;
            // F000:0293 add DX,CX
            DX = (ushort)(DX + CX);
            // F000:0295 mov word ptr DS:[BX+0x2E],DX
            UInt16[DS, (ushort)(BX + 0x2E)] = DX;
            // F000:0298 mov DX,AX
            DX = AX;
            // F000:029A sub DX,CX
            DX = (ushort)(DX - CX);
            // F000:029C mov word ptr DS:[BX+0x30],DX
            UInt16[DS, (ushort)(BX + 0x30)] = DX;
            // F000:029F mov DX,AX
            DX = AX;
            // F000:02A1 inc DX
            DX++;
            // F000:02A2 mov word ptr DS:[BX+0x32],DX
            UInt16[DS, (ushort)(BX + 0x32)] = DX;
            // F000:02A5 mov DX,CX
            DX = CX;
            // F000:02A7 dec DX
            DX--;
            // F000:02A8 mov word ptr DS:[BX+0x34],DX
            UInt16[DS, (ushort)(BX + 0x34)] = DX;
            // F000:02AB mov DX,AX
            DX = AX;
            // F000:02AD neg DX
            DX = (ushort)-DX;
            CarryFlag = DX != 0;
            // F000:02AF mov word ptr DS:[BX+0x36],DX
            UInt16[DS, (ushort)(BX + 0x36)] = DX;
            // F000:02B2 mov DX,AX
            DX = AX;
            // F000:02B4 or DX,CX
            DX = (ushort)(DX | CX);
            // F000:02B6 mov word ptr DS:[BX+0x38],DX
            UInt16[DS, (ushort)(BX + 0x38)] = DX;
            // F000:02B9 mov byte ptr DS:[BX+0x3A],AL
            UInt8[DS, (ushort)(BX + 0x3A)] = AL;
            // F000:02BC add byte ptr DS:[BX+0x3A],5
            UInt8[DS, (ushort)(BX + 0x3A)] = (byte)(UInt8[DS, (ushort)(BX + 0x3A)] + 5);
            // F000:02C0 mov DI,AX
            DI = AX;
            // F000:02C2 add DI,AX
            DI = Alu16.Add(DI, AX);
            // F000:02C4 mov DX,0
            DX = 0;
            // F000:02C7 adc DL,0
            DL = (byte)(DL + (CarryFlag ? 1 : 0));
            // F000:02CA mov byte ptr DS:[BX+0x3B],DL
            UInt8[DS, (ushort)(BX + 0x3B)] = DL;
            // F000:02CD mov DX,AX
            DX = AX;
            // F000:02CF inc DX
            DX++;
            // F000:02D0 inc DX
            DX++;
            // F000:02D1 mov byte ptr DS:[BX+0x3C],DL
            UInt8[DS, (ushort)(BX + 0x3C)] = DL;
            // F000:02D4 cmp AX,CX
            // F000:02D6 test AX,AX
            // F000:02D8 mov byte ptr DS:[BX+0x3D],7
            UInt8[DS, (ushort)(BX + 0x3D)] = 7;
            // F000:02DC cmp AX,word ptr DS:[0x0100]
            Alu16.Sub(AX, UInt16[DS, 0x0100]);
            // F000:02E0 mov byte ptr DS:[BX+0x3E],9
            UInt8[DS, (ushort)(BX + 0x3E)] = 9;
            // F000:02E4 mov DX,BP
            DX = BP;
            // F000:02E6 mov byte ptr DS:[BX+0x3F],DL
            UInt8[DS, (ushort)(BX + 0x3F)] = DL;
            // F000:02E9 add BX,0x0040
            BX = (ushort)(BX + 0x0040);
            // F000:02EC add SI,4
            SI = Alu16.Add(SI, 4);
            // F000:02EF dec BP
            BP = Alu16.Dec(BP);
            // F000:02F0 jne near 0x0018
        } while (BP != 0);
        // F000:02F4 hlt
        return Hlt();
    }

        VerifySpeculativeEntryOrFail(biosSegment, 0x00F2, Signature_00F2_callback);
        // F000:00F2 callback 0x010D
        // Spice86 callback: MouseDriver.AfterUserHandlerExecution
        Callback(biosSegment, 0x00F2, 0x010D);
        VerifySpeculativeEntryOrFail(biosSegment, 0x00F6, Signature_00F6_ret);
        // F000:00F6 ret far
        return FarRet();
    }

}

                Alu16.Sub(CX, 3);

                    AX = 0xF000;

                        break;
                    }
                    // F000:101C jmp near word ptr ES:[SI+0x1035]

                            goto L_1027;
                        default:
                            throw FailAsUntested($"Untested near jump target 0x{UInt16[ES, (ushort)(SI + 0x1035)]:X4} at F000:101C");
                    }
                }
                // F000:1025 jmp short 0x1029
                // F000:1029 inc word ptr DS:[0x0040]
                Segment2000Globals.Word0040 += 1;
                // F000:102D jmp short 0x1021

public partial class CfgGeneratedOverrides : CSharpOverrideHelper {
    protected readonly ushort biosSegment;

    private readonly Func<int, Action> unknown_F000_000F_F000FEntry;
    private readonly Func<int, Action> unknown_F000_0014_F0014Entry;
    private readonly Func<int, Action> unknown_F000_FFF0_FFFF0Entry;
    private readonly Func<int, Action> unknown_F000_0019_F0019_AsmEntry;

    public CfgGeneratedOverrides(IDictionary<SegmentedAddress, FunctionInformation> functionInformations, Machine machine, ILogger loggerService, Configuration configuration, ushort programStartSegment) : base(functionInformations, machine, loggerService, configuration) {
        biosSegment = 0xF000;

        unknown_F000_000F_F000FEntry = _ => unknown_F000_000F_F000F();
        unknown_F000_0014_F0014Entry = _ => unknown_F000_0014_F0014();
        unknown_F000_FFF0_FFFF0Entry = _ => unknown_F000_FFF0_FFFF0();
        unknown_F000_0019_F0019_AsmEntry = _ => unknown_F000_0019_F0019_Asm();
        DefineFunctionWithoutInterpreterFallback(biosSegment, 0x000F, unknown_F000_000F_F000FEntry, "unknown");
        DefineFunctionWithoutInterpreterFallback(biosSegment, 0x0014, unknown_F000_0014_F0014Entry, "unknown");
        DefineFunctionWithoutInterpreterFallback(biosSegment, 0xFFF0, unknown_F000_FFF0_FFFF0Entry, "unknown");
        DefineFunctionWithoutInterpreterFallback(biosSegment, 0x0019, unknown_F000_0019_F0019_AsmEntry, "unknown");
    }

    protected override Func<int, Action> HardwareInterruptFunction(byte vectorNumber, SegmentedAddress target) =>
        (vectorNumber, target.Segment, target.Offset) switch {
            _ => throw FailAsUntested($"Hardware interrupt 0x{vectorNumber:X2} targets {target}, which discovery did not observe as a handler of that vector")
        };

    protected override bool ObservedCallLandingsOnly => true;
}

using Spice86.Core.CLI;
using Spice86.Core.Emulator.CPU.Exceptions;
using Spice86.Core.Emulator.Function;
using Spice86.Core.Emulator.ReverseEngineer;
using Spice86.Core.Emulator.VM;
using Microsoft.Extensions.Logging;

using System.Collections.Generic;

namespace Spice86.Generated;

public partial class CfgGeneratedOverrides {
    /// <summary>Inputs: DS. Does not return.</summary>
    public virtual Action unknown_F000_FFF0_FFFF0() {
        byte al1 = 0;
        CheckExternalEvents(biosSegment, 0xFFF0);
        // F000:FFF0 jmp short 0
        // F000:0000 mov byte ptr DS:[0x0400],1
        DsGlobals.Byte0400 = 1;
        // F000:0005 mov AL,byte ptr DS:[0x0500]
        al1 = DsGlobals.Byte0500;
        // F000:0008 test AL,AL
        Alu8.And(al1, al1);
        // F000:000A jne short 0x0013
        if (al1 != 0) {
            throw FailAsUntested("Untested jump at F000:000A");
        }
        // F000:000C mov byte ptr DS:[0x0401],0xDD
        DsGlobals.Byte0401 = 0xDD;
        // F000:0011 jmp short 0x0017
        // F000:0017 mov byte ptr DS:[0x0402],0xEE
        DsGlobals.Byte0402 = 0xEE;
        // F000:001C hlt
        AL = al1;
        return Hlt();
    }

}

using Spice86.Core.Emulator.CPU.Registers;
using Spice86.Core.Emulator.Memory;
using Spice86.Core.Emulator.ReverseEngineer.DataStructure;

namespace Spice86.Generated;

public sealed class DsGlobals : MemoryBasedDataStructureWithDsBaseAddress {
    private readonly IMemory _memory;
    private readonly SegmentRegisters _segmentRegisters;

    private ushort Segment => _segmentRegisters.UInt16[(uint)SegmentRegisterIndex.DsIndex];

    public DsGlobals(IMemory memory, SegmentRegisters segmentRegisters) : base(memory, segmentRegisters) {
        _memory = memory;
        _segmentRegisters = segmentRegisters;
    }

    /// DS:0x0400, whatever DS holds at the access.
    public byte Byte0400 { get => _memory.UInt8[Segment, 0x0400]; set => _memory.UInt8[Segment, 0x0400] = value; }

    /// DS:0x0401, whatever DS holds at the access.
    public byte Byte0401 { get => _memory.UInt8[Segment, 0x0401]; set => _memory.UInt8[Segment, 0x0401] = value; }

    /// DS:0x0402, whatever DS holds at the access.
    public byte Byte0402 { get => _memory.UInt8[Segment, 0x0402]; set => _memory.UInt8[Segment, 0x0402] = value; }

    /// DS:0x0500, whatever DS holds at the access.
    public byte Byte0500 { get => _memory.UInt8[Segment, 0x0500]; set => _memory.UInt8[Segment, 0x0500] = value; }
}

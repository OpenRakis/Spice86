using Spice86.Core.Emulator.CPU.Registers;
using Spice86.Core.Emulator.Memory;
using Spice86.Core.Emulator.ReverseEngineer.DataStructure;

namespace Spice86.Generated;

public sealed class DsGlobals : MemoryBasedDataStructureWithDsBaseAddress {
    private readonly IMemory _memory;
    private readonly SegmentRegisters _segmentRegisters;

    private ushort Segment => _segmentRegisters.UInt16[(uint)SegmentRegisterIndex.DsIndex];

    public DsGlobals(IMemory memory, SegmentRegisters segmentRegisters) : base(memory, segmentRegisters) {
        _memory = memory;
        _segmentRegisters = segmentRegisters;
    }

    /// DS:0x0020, whatever DS holds at the access.
    public ushort Word0020 { get => _memory.UInt16[Segment, 0x0020]; set => _memory.UInt16[Segment, 0x0020] = value; }

    /// DS:0x0022, whatever DS holds at the access.
    public ushort Word0022 { get => _memory.UInt16[Segment, 0x0022]; set => _memory.UInt16[Segment, 0x0022] = value; }
}
